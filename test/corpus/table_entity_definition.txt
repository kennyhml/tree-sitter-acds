==================
Logical and relational condition variants
==================

define table entity condition_variants {
  id : abap.int4;

  _comparison : association to ComparisonTarget on
       _comparison.equal = $projection.id
    and _comparison.unequal <> $projection.id
    and _comparison.less < $projection.id
    and _comparison.greater > $projection.id
    and _comparison.less_equal <= $projection.id
    and _comparison.greater_equal >= 0;

  _logic : association to LogicTarget on
    not (
      _logic.rank between 1 and 3
      or _logic.score not between -10 and +10
    )
    and _logic.name like 'A%'
    and _logic.code not like 'A#_%' escape '#'
    and _logic.nullable is null
    and _logic.required is not null
    and _logic.empty is initial
    and _logic.filled is not initial
    and _logic.language = $session.system_language
    and _logic.status = #OPEN
    and _logic.created = abap.dats'20260906'
    with default filter
      active = 'X' or tenant = $session.client;
}
