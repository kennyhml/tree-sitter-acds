==================
Parameters in both BETWEEN bounds and LIKE pattern
==================

view entity FilteredView
  with parameters
    p_min : AmountType,
    p_max : AmountType,
    p_pattern : PatternType
  as select from Source { id }
  where amount between $parameters.p_min and $parameters.p_max
    and name like $parameters.p_pattern

==================
Parameter in association ON and default filter
==================

view entity AssociationView
  with parameters p_id : IdType
  as select from Source
    association to Target as _target on _target.id = $parameters.p_id
      with default filter id = $parameters.p_id
  { id, _target }
